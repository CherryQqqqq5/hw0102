    23	template<typename T>
    24	static tree_ptr<T> make_tree(const std::vector<T> &data)
    25	{
    26	    if (data.empty())
    27	        return tree_ptr<T>(nullptr);
    28	    
    29	    auto data_iter = data.begin();
    30	    auto tree = tree_ptr<T>(new binary_search_tree::binary_tree<T>(*data_iter));
    31	    ++data_iter;
    32	
    33	    for (; data_iter != data.end(); ++data_iter)
    34	    {
    35	        tree->insert(*data_iter);
    36	    }
    37	
    38	    return tree;
    39	}
